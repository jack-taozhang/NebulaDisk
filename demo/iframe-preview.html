// /onlinePreview 的 url 参数 = Base64(文件地址) 之后再 encodeURIComponent
function b64utf8(str) {
  const bytes = new TextEncoder().encode(str);
  let bin = '';
  bytes.forEach(b => bin += String.fromCharCode(b));
  return btoa(bin);
}

function previewUrl(fileUrl, base = '/kkfileview') {
  return base + '/onlinePreview?url=' + encodeURIComponent(b64utf8(fileUrl));
}
